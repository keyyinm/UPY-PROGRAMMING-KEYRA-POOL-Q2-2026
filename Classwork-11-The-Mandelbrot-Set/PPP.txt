# Classwork 11 - The Mandelbrot Set

# Read configuration values

Create empty dictionary called config

Open config.txt for reading

For each line in config.txt

    Split line using "="

    Store key and value in config

End For

Close config.txt

# Get parameters

ancho ← value of ancho from config

alto ← value of alto from config

max_iter ← value of max_iter from config

# Create output file

Open mandelbrot.csv for writing

Write "fila,columna,iteraciones" to output file

# Process all points in the grid

For fila ← 0 to alto - 1

    For columna ← 0 to ancho - 1

        real ← real_min + (columna ÷ ancho) × (real_max - real_min)

        imag ← imag_min + (fila ÷ alto) × (imag_max - imag_min)

        c ← complex number using real and imag

        z ← 0

        iteraciones ← 0

        While magnitude of z ≤ 2 AND iteraciones < max_iter

            z ← z × z + c

            iteraciones ← iteraciones + 1

        End While

        Write fila, columna, iteraciones to output file

    End For

End For

Close output file

Display "Mandelbrot done"
